โค้ดภาษา C: ระบบจัดการคลังสินค้า (inventory_core.c) 

/*
 * Computer Hardware Inventory Core Engine in C
 * Author: TechStock Architecture
 * Compile command for WebAssembly:
 * emcc inventory_core.c -O3 -s WASM=1 -s EXPORTED_FUNCTIONS="['_stockIn', '_stockOut', '_checkLowStock']" -o inventory.js
 */

#include <stdio.h>
#include <string.h>
#include <stdbool.h>

#define MAX_ITEMS 100
#define ALERT_THRESHOLD_DEFAULT 5

// โครงสร้างข้อมูลสินค้าคอมพิวเตอร์
typedef struct {
    int id;
    char sku[20];
    char name[100];
    char category[30]; // CPU, GPU, RAM, SSD, Mainboard, PSU
    int quantity;
    int min_threshold;
    double price;
} HardwareItem;

HardwareItem inventory[MAX_ITEMS];
int item_count = 0;

// ฟังก์ชันนำเข้าสินค้า (Stock In)
bool stockIn(int item_id, int qty) {
    for (int i = 0; i < item_count; i++) {
        if (inventory[i].id == item_id) {
            inventory[i].quantity += qty;
            printf("[SUCCESS] Stock In ID %d: +%d units. New Total: %d\n", item_id, qty, inventory[i].quantity);
            return true;
        }
    }
    printf("[ERROR] Item ID %d not found!\n", item_id);
    return false;
}

// ฟังก์ชันนำออกสินค้า (Stock Out) พร้อมตรวจสอบสินค้าหมด
bool stockOut(int item_id, int qty) {
    for (int i = 0; i < item_count; i++) {
        if (inventory[i].id == item_id) {
            if (inventory[i].quantity < qty) {
                printf("[ALERT] Insufficient stock for ID %d! Available: %d, Requested: %d\n", item_id, inventory[i].quantity, qty);
                return false;
            }
            inventory[i].quantity -= qty;
            printf("[SUCCESS] Stock Out ID %d: -%d units. Remaining: %d\n", item_id, qty, inventory[i].quantity);
            
            // แจ้งเตือนเมื่อใกล้หมด
            if (inventory[i].quantity <= inventory[i].min_threshold) {
                printf("[WARNING] Low Stock Alert for %s (ID: %d)! Only %d left.\n", inventory[i].name, item_id, inventory[i].quantity);
            }
            return true;
        }
    }
    return false;
}

// ตรวจสอบสินค้าที่ต่ำกว่าเกณฑ์เตือน
void checkLowStockAlerts() {
    printf("\n=== CRITICAL LOW STOCK REPORT ===\n");
    int alert_count = 0;
    for (int i = 0; i < item_count; i++) {
        if (inventory[i].quantity <= inventory[i].min_threshold) {
            printf("- [SKU: %s] %s | Current: %d (Min Limit: %d)\n", 
                   inventory[i].sku, inventory[i].name, inventory[i].quantity, inventory[i].min_threshold);
            alert_count++;
        }
    }
    if(alert_count == 0) printf("All hardware stocks are sufficient.\n");
}

int main() {
    printf("Hardware Inventory C Engine Initialized.\n");
    return 0;
}

ขั้นตอนการ Publish ขึ้น GitHub Pages

  1. สร้าง Repository ใหม่ใน GitHub ตั้งชื่อ เช่น hardware-inventory-system
  2. อัปโหลดไฟล์ index.html นี้ไว้ที่ Root Directory ของ Repository
  3. ไปที่เมนู **Settings** -> **Pages** ใน Repository ของคุณ
  4. ตรงหัวข้อ **Source** เลือก **Deploy from a branch** แล้วเลือก Branch main (หรือ master) / root แล้วกด **Save**
  5. รอประมาณ 1-2 นาที ระบบจะมอบ URL ให้ เช่น
    https://yourusername.github.io/hardware-inventory-system/

CI/CD ด้วย GitHub Actions (อัตโนมัติ)

หากต้องการคอมไพล์ C Code อัตโนมัติด้วย Emscripten แล้ว Deploy WebAssembly ลง GitHub Pages ให้สร้างไฟล์ .github/workflows/deploy.yml: 

name: Deploy Inventory WebAssembly
on:
  push:
    branches: [ main ]
jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Setup Emscripten
        uses: mymindstorm/setup-emsdk@v12
      - name: Compile C to WASM
        run: emcc inventory_core.c -O3 -o build/inventory.js
      - name: Deploy Pages
        uses: peaceiris/actions-gh-pages@v3
        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}
          publish_dir: ./build
